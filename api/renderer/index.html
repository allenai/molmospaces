    Filament keeps its own retained scene (ModelRenderables/ModelLights), which update() syncs from MjData. An MjvScene is still filled on update() so callers can read or override scene.camera before render(), as they do with the OpenGL renderer.

    def __init__(
    self,
    model: mj.MjModel,
    device_id: int | None = None,
    height: int = 720,
    width: int = 1280,
    max_geom: int = 10000,
) -> None:
    if mjrf is None:
        raise ImportError(
            "Filament rendering requires mujoco>=4.0 with the `mujoco-studio` package."
        )
    super().__init__(model, device_id)

    self._width = width
    self._height = height

    self._scene = mj.MjvScene(model=model, maxgeom=max_geom)
    self._scene_option = mj.MjvOption()
    self._scene_option.sitegroup *= 0

    self._depth_rendering = False
    self._segmentation_rendering = False

    self._ctx = mjrf.Context(mjrf.ContextConfig())
    self._ctx.set_clear_color([0.0, 0.0, 0.0])
    self._resources = mjrf.ModelResources(self._ctx, model)
    self._fscene = self._ctx.create_scene(mjrf.SceneParams())
    self._fscene.configure_from_model(model)
    self._lights = mjrf.ModelLights(self._fscene, self._resources)
    self._renderables = mjrf.ModelRenderables(self._fscene, self._resources)
    self._renderables.set_options(self._scene_option)
    self._headlight_intensity = self._get_headlight_intensity(model)

    # (pixel_format, width, height) -> (RenderTarget, ReadRequest, buffer)
    self._targets: dict[tuple, tuple] = {}

    def render(
    self,
    *,
    out: np.ndarray | None = None,
    width: int | None = None,
    height: int | None = None,
) -> np.ndarray:
    height = height or self._height
    width = width or self._width

    if self._depth_rendering:
        out_shape = (height, width)
        draw_mode = mjrf.DrawMode.DRAW_MODE_DEPTH
        target, read, buffer = self._target(
            mjrf.PixelFormat.PIXEL_FORMAT_R32F, width, height, 1, np.float32
        )
    else:
        out_shape = (height, width, 3)
        draw_mode = (
            mjrf.DrawMode.DRAW_MODE_SEGMENTATION_BY_ID
            if self._segmentation_rendering
            else mjrf.DrawMode.DRAW_MODE_DEFAULT
        )
        target, read, buffer = self._target(
            mjrf.PixelFormat.PIXEL_FORMAT_RGB8, width, height, 3, np.uint8
        )

    if out is not None and out.shape != out_shape:
        raise ValueError(
            f"Expected `out.shape == {out_shape}`. Got `out.shape={out.shape}`"
            " instead. When using depth rendering, the out array should be of"
            " shape `(height, width)` and otherwise (height, width, 3)."
            f" Got `self._depth_rendering={self._depth_rendering}`."
        )

    # Depth and segmentation must not be tone mapped or dithered.
    plain = self._depth_rendering or self._segmentation_rendering
    request = mjrf.RenderRequest()
    request.scene = self._fscene
    request.target = target
    request.camera = self._camera()
    request.viewport.left = 0
    request.viewport.bottom = 0
    request.viewport.width = width
    request.viewport.height = height
    request.draw_mode = draw_mode
    request.post_processing = not plain
    request.shadows = True
    request.reflections = True
    request.skybox = True
    request.haze = True
    request.headlight = self.model.vis.headlight.active != 0
    request.headlight_color = tuple(self.model.vis.headlight.diffuse)
    request.headlight_intensity = self._headlight_intensity

    self._ctx.wait_for_frame(self._ctx.render([request], [read]))

    if self._depth_rendering:
        # Filament writes reverse-Z depth with a finite far plane:
        #   z = near * (far - d) / (d * (far - near))
        gl = self._scene.camera[0]
        near = np.float64(gl.frustum_near)
        far = np.float64(gl.frustum_far)
        z = buffer.astype(np.float64)
        if gl.orthographic:
            depth = far - z * (far - near)
        else:
            depth = near * far / (z * (far - near) + near)
        result = depth.astype(np.float32)
    elif self._segmentation_rendering:
        result = self._decode_segmentation(buffer)
    else:
        result = buffer

    if out is None:
        return result.copy() if result is buffer else result
    out[:] = result
    return out

    def update(
    self,
    data: mj.MjData,
    camera: int | str | mj.MjvCamera = -1,
    scene_option: mj.MjvOption | None = None,
) -> None:
    if not isinstance(camera, mj.MjvCamera):
        camera_id = camera
        if isinstance(camera_id, str):
            camera_id = mj.mj_name2id(self.model, mj.mjtObj.mjOBJ_CAMERA.value, camera_id)
            if camera_id == -1:
                raise ValueError(f'The camera "{camera}" does not exist.')
        if camera_id < -1 or camera_id >= self.model.ncam:
            raise ValueError(
                f"The camera id {camera_id} is out of range [-1, {self.model.ncam})."
            )

        camera = mj.MjvCamera()
        camera.fixedcamid = camera_id

        if camera_id == -1:
            camera.type = mj.mjtCamera.mjCAMERA_FREE
            mj.mjv_defaultFreeCamera(self.model, camera)
        else:
            camera.type = mj.mjtCamera.mjCAMERA_FIXED

    scene_option = scene_option or self._scene_option
    # Fills scene.camera, which callers may override before render().
    mj.mjv_updateScene(
        self.model,
        data,
        scene_option,
        None,
        camera,
        mj.mjtCatBit.mjCAT_ALL.value,
        self._scene,
    )
    self._renderables.set_options(scene_option)
    self._lights.update(data)
    self._renderables.update(data)

    def create_initialized_egl_device_display(device_id=0):
    """Creates an initialized EGL display directly on a device."""
    all_devices = EGL.eglQueryDevicesEXT()
    selected_device = (
        os.environ.get("CUDA_VISIBLE_DEVICES", None)
        if os.environ.get("MUJOCO_EGL_DEVICE_ID", None) is None
        else os.environ.get("MUJOCO_EGL_DEVICE_ID", None)
    )
    if selected_device is None:
        candidates = all_devices
        device_idx = 0 if device_id == -1 else device_id
    else:
        if not selected_device.isdigit():
            device_inds = [int(x) for x in selected_device.split(",")]
            if device_id == -1:
                device_idx = device_inds[0]
            else:
                assert device_id in device_inds, (
                    "specified device id is not made visible in environment variables."
                )
                device_idx = device_id
        else:
            device_idx = int(selected_device)
        if not 0 <= device_idx < len(all_devices):
            raise RuntimeError(
                f"The MUJOCO_EGL_DEVICE_ID environment variable must be an integer "
                f"between 0 and {len(all_devices) - 1} (inclusive), got {device_idx}."
            )
    candidates = all_devices[device_idx : device_idx + 1]
    for device in candidates:
        display = EGL.eglGetPlatformDisplayEXT(EGL.EGL_PLATFORM_DEVICE_EXT, device, None)
        if display != EGL.EGL_NO_DISPLAY and EGL.eglGetError() == EGL.EGL_SUCCESS:
            # `eglInitialize` may or may not raise an exception on failure depending
            # on how PyOpenGL is configured. We therefore catch a `GLError` and also
            # manually check the output of `eglGetError()` here.
            try:
                initialized = EGL.eglInitialize(display, None, None)
            except error.GLError:
                pass
            else:
                if initialized == EGL.EGL_TRUE and EGL.eglGetError() == EGL.EGL_SUCCESS:
                    return display
    return EGL.EGL_NO_DISPLAY

    def render(
    self,
    *,
    out: np.ndarray | None = None,
    width: int | None = None,
    height: int | None = None,
) -> np.ndarray:
    """Renders the scene as a numpy array of pixel values.

    Args:
      out: Alternative output array in which to place the resulting pixels. It
        must have the same shape as the expected output but the type will be
        cast if necessary. The expted shape depends on the value of
        `self._depth_rendering`: when `True`, we expect `out.shape == (width,
        height)`, and `out.shape == (width, height, 3)` when `False`.

    Returns:
      A new numpy array holding the pixels with shape `(H, W)` or `(H, W, 3)`,
      depending on the value of `self._depth_rendering` unless
      `out is None`, in which case a reference to `out` is returned.

    Raises:
      RuntimeError: if this method is called after the close method.
    """
    assert self._scene is not None, "Internal scene:MjvScene must be initialized by now"

    height = height or self._height
    width = width or self._width
    rect = mj.MjrRect(0, 0, width, height)

    original_flags = self._scene.flags.copy()

    # Enable shadow rendering (required for shadows to appear in rendered images)
    # Shadows are controlled by lights with castshadow enabled
    self._scene.flags[mj.mjtRndFlag.mjRND_SHADOW] = True

    # Using segmented rendering for depth makes the calculated depth more
    # accurate at far distances.
    if self._depth_rendering or self._segmentation_rendering:
        self._scene.flags[mj.mjtRndFlag.mjRND_SEGMENT] = True
        self._scene.flags[mj.mjtRndFlag.mjRND_IDCOLOR] = True

    if self._gl_context is None:
        raise RuntimeError("render cannot be called after close.")

    self._gl_context.make_current()

    # Upload textures to GPU before rendering if textures have been modified
    # This is necessary when textures are modified via model.tex_data
    # Only upload when needed to avoid performance overhead
    if self._textures_need_upload:
        self.upload_textures()
        self._textures_need_upload = False

    if self._depth_rendering:
        out_shape = (rect.height, rect.width)
        out_dtype = np.float32
    else:
        out_shape = (rect.height, rect.width, 3)
        out_dtype = np.uint8

    if out is None:
        out = np.empty(out_shape, dtype=out_dtype)
    else:
        if out.shape != out_shape:
            raise ValueError(
                f"Expected `out.shape == {out_shape}`. Got `out.shape={out.shape}`"
                " instead. When using depth rendering, the out array should be of"
                " shape `(width, height)` and otherwise (width, height, 3)."
                f" Got `(self.height, self.width)={(self.height, self.width)}` and"
                f" `self._depth_rendering={self._depth_rendering}`."
            )

    assert self._mjr_context, "MjrContext must be created by now, but it's None"
    mj.mjr_render(rect, self._scene, self._mjr_context)

    if self._depth_rendering:
        mj.mjr_readPixels(rgb=None, depth=out, viewport=rect, con=self._mjr_context)

        # Get the distances to the near and far clipping planes.
        extent = self.model.stat.extent
        near = self.model.vis.map.znear * extent
        far = self.model.vis.map.zfar * extent

        # Calculate OpenGL perspective matrix values in float32 precision
        # so they are close to what glFrustum returns
        # https://registry.khronos.org/OpenGL-Refpages/gl2.1/xhtml/glFrustum.xml
        zfar = np.float32(far)
        znear = np.float32(near)
        c_coef = -(zfar + znear) / (zfar - znear)
        d_coef = -(np.float32(2) * zfar * znear) / (zfar - znear)

        # In reverse Z mode the perspective matrix is transformed by the following
        c_coef = np.float32(-0.5) * c_coef - np.float32(0.5)
        d_coef = np.float32(-0.5) * d_coef

        # We need 64 bits to convert Z from ndc to metric depth without noticeable
        # losses in precision
        out_64 = out.astype(np.float64)

        # Undo OpenGL projection
        # Note: We do not need to take action to convert from window coordinates
        # to normalized device coordinates because in reversed Z mode the mapping
        # is identity
        out_64 = d_coef / (out_64 + c_coef)

        # Cast result back to float32 for backwards compatibility
        # This has a small accuracy cost
        out[:] = out_64.astype(np.float32)

        # Reset scene flags.
        np.copyto(self._scene.flags, original_flags)
    elif self._segmentation_rendering:
        mj.mjr_readPixels(rgb=out, depth=None, viewport=rect, con=self._mjr_context)

        # Convert 3-channel uint8 to 1-channel uint32.
        image3 = out.astype(np.uint32)
        segimage = image3[:, :, 0] + image3[:, :, 1] * (2**8) + image3[:, :, 2] * (2**16)
        # Remap segid to 3-channel (object ID, object type, body ID) triplet
        # Seg ID 0 is background -- will be remapped to (-1, -1, -1).

        # Find the maximum segment ID in the image to size the output array correctly
        max_segid = np.max(segimage) if segimage.size > 0 else 0

        # Create output array with size to accommodate all possible segment IDs
        # Add 1 to account for 0-based indexing and ensure we have enough space
        segid2output = np.full((max_segid + 1, 3), fill_value=-1, dtype=np.int32)

        visible_geoms = [g for g in self._scene.geoms[: self._scene.ngeom] if g.segid != -1]
        visible_segids = np.array([g.segid + 1 for g in visible_geoms], np.int32)
        visible_objid = np.array([g.objid for g in visible_geoms], np.int32)
        visible_objtype = np.array([g.objtype for g in visible_geoms], np.int32)
        visible_bodyid = np.array(
            [self.geomid_to_bodyid(g.objid) for g in visible_geoms], np.int32
        )

        # Only set values for valid segment IDs that are within bounds
        valid_mask = (visible_segids >= 0) & (visible_segids < segid2output.shape[0])
        if np.any(valid_mask):
            segid2output[visible_segids[valid_mask], 0] = visible_objid[valid_mask]
            segid2output[visible_segids[valid_mask], 1] = visible_objtype[valid_mask]
            segid2output[visible_segids[valid_mask], 2] = visible_bodyid[valid_mask]

        out = segid2output[segimage]

        # Reset scene flags.
        np.copyto(self._scene.flags, original_flags)
    else:
        mj.mjr_readPixels(rgb=out, depth=None, viewport=rect, con=self._mjr_context)
        mj.mjr_readPixels(rgb=out, depth=None, viewport=rect, con=self._mjr_context)

    out[:] = np.flipud(out)

    # TODO In MacOS, keeping the context locked seems to preclude others to progress,
    #  so it doesn't look like we can achieve true parallelism through multi threading?

    pbr_materials ¶

    PBR material overrides for assets that ship without PBR parameters.

    Almost none of our assets carry PBR material data. Every objaverse object MJCF declares exactly one material with nothing but an albedo texture, and the THOR scene materials are the same story, so mat_metallic/mat_roughness stay at their "unset" sentinel of -1 and the whole scene is shaded by one undifferentiated legacy response. This module fills that gap at load time, from the material composition already recorded in the objathor metadata.

    What Filament's PBR shader (pbr.filamat, shipped with MuJoCo 4.0 in mujoco/studio/assets) actually does with the numbers we set -- this drives every decision below::

    material.roughness  = RoughnessFactor;                 // <- mat_roughness
material.roughness *= texture(Roughness, uv).x;        // <- mjTEXROLE_ROUGHNESS
material.metallic   = MetallicFactor;                  // <- mat_metallic
material.metallic  *= texture(Metallic, uv).x;         // <- mjTEXROLE_METALLIC
...
diffuseColor = baseColor * (1 - metallic);
f0           = baseColor * metallic + reflectance * (1 - metallic);
computeDielectricF0(reflectance) = 0.16 * reflectance * reflectance;

    Three consequences worth stating, because they are not what the MuJoCo docs imply:

    1. Factors multiply their texture (the glTF convention), they do not replace it. So the per-class roughness below is a factor and the noise map is authored with a mean of 1.0 -- see :func:roughness_noise_texture.
    2. metallic = 1 zeroes the diffuse term and makes f0 the base colour, which is exactly the "baseColor dictates the colour of the metal reflection" rule.
    3. There is no Reflectance uniform in the shipped material -- MaterialParams holds only BaseColorFactor, MetallicFactor and RoughnessFactor. So reflectance keeps Filament's default of 0.5, i.e. an f0 of 0.16 * 0.5^2 = 0.04, the 4% Fresnel interface we want for dielectrics. We get the dielectric reflectance rule for free and cannot change it from MJCF; MuJoCo's mat_reflectance does not reach this shader.

    """
    materials = list(spec.materials)
    stats = PbrOverrideStats(total=len(materials))

    opaque = [material for material in materials if not _is_transparent(material)]
    stats.skipped_transparent = len(materials) - len(opaque)

    pending = [material for material in opaque if not _has_authored_pbr(material)]
    stats.skipped_authored = len(opaque) - len(pending)
    if not pending:
        return stats

    if annotations is None:
        from molmo_spaces.utils.object_metadata import ObjectMeta

        annotations = ObjectMeta.annotation()

    # The shader multiplies MetallicFactor by the metallic texture unconditionally.
    # We bind an explicit all-ones map rather than trusting whatever MuJoCo binds
    # when the role is empty -- if that default were not white, every metal would
    # silently come out as a dielectric. 2x2 because MuJoCo 4.0's Filament upload
    # throws on a 1x1 single-channel texture (2x2 and larger are fine).
    _add_data_texture(spec, UNIT_TEXTURE, np.full((2, 2), 255, dtype=np.uint8))

    # Roughness maps live in UV space at full extent so they need no tiling (material
    # texrepeat applies to every layer at once, albedo included, so a tiled detail map
    # is not an option here). One texture is added per finish that actually gets used,
    # and every material that draws that finish shares it.
    added: dict[str, str] = {}

    def texture_for(finish: str) -> str:
        if finish not in added:
            name = f"{ROUGHNESS_TEXTURE_PREFIX}{finish}"
            _add_data_texture(
                spec, name, roughness_texture(finish, noise_resolution, noise_amplitude, seed)
            )
            added[finish] = name
        return added[finish]

    added_normals: dict[str, str] = {}

    # Albedo filenames, so a material can be matched to the authored maps beside it.
    albedo_files = {texture.name: texture.file for texture in spec.textures}
    authored = _authored_map_index() if use_authored_maps else {}
    authored_added: dict[tuple[str, str], str] = {}

    def authored_texture(stem: str, kind: str, data: np.ndarray) -> str:
        key = (stem, kind)
        if key not in authored_added:
            name = f"{AUTHORED_TEXTURE_PREFIX}{kind}_{stem}"[:90]
            _add_data_texture(spec, name, data)
            authored_added[key] = name
        return authored_added[key]

    def maps_for(material) -> dict[str, Path]:
        albedo = (
            material.textures[int(mj.mjtTextureRole.mjTEXROLE_RGB)]
            or material.textures[int(mj.mjtTextureRole.mjTEXROLE_RGBA)]
        )
        if not albedo:
            return {}
        source = albedo_files.get(albedo)
        return (authored.get(albedo_stem(source)) or {}) if source else {}

    def normal_for(finish: str) -> str:
        if finish not in added_normals:
            name = f"{NORMAL_TEXTURE_PREFIX}{finish}"
            _add_data_texture(
                spec, name, normal_texture(finish, noise_resolution, bump_strength, seed)
            )
            added_normals[finish] = name
        return added_normals[finish]

    # Seeded separately from the maps themselves, so which finish a material draws is
    # reproducible for a scene but independent of what the maps look like.
    chooser = random.Random(seed)
    gain = roughness_factor_gain(noise_amplitude)
    for material in pending:
        pbr_class, source = _classify(material.name, annotations)
        finish = chooser.choice(pbr_class.finishes)

        real = maps_for(material)

        material.metallic = pbr_class.metallic
        # min() guards the encoding: the gain is undone by the map's mean, but the
        # factor itself still has to be a legal [0, 1] roughness. Where it bites, the
        # effective roughness is compressed to 1/gain instead of the class value --
        # at the default amplitude that is anything above ~0.77, i.e. only the matte
        # end (fabric, plaster, brick), where roughness barely reads anyway. The
        # alternative, a differently-scaled map per roughness value, would multiply
        # the number of 1024-square textures for no visible gain.
        material.roughness = min(1.0, pbr_class.roughness * gain)
        material.textures[int(mj.mjtTextureRole.mjTEXROLE_ROUGHNESS)] = texture_for(finish)
        material.textures[int(mj.mjtTextureRole.mjTEXROLE_METALLIC)] = UNIT_TEXTURE

        # An authored metallic/smoothness map replaces both guesses outright: it says
        # per texel what the surface is, so the factors become plain 1.0 pass-throughs.
        packed = (
            _load_map(real["metallic_smoothness"], noise_resolution)
            if ("metallic_smoothness" in real)
            else None
        )
        if packed is not None:
            stem = albedo_stem(
                albedo_files.get(material.textures[int(mj.mjtTextureRole.mjTEXROLE_RGB)], "")
            )
            metallic_map, roughness_map = split_metallic_smoothness(packed)
            material.metallic = 1.0
            material.roughness = 1.0
            material.textures[int(mj.mjtTextureRole.mjTEXROLE_METALLIC)] = authored_texture(
                stem, "metallic", metallic_map
            )
            material.textures[int(mj.mjtTextureRole.mjTEXROLE_ROUGHNESS)] = authored_texture(
                stem, "roughness", roughness_map
            )
            stats.authored_metallic_roughness += 1

        # The procedural relief comes from the same finish as the roughness, so the two
        # stay in register rather than reading as unrelated patterns; an authored normal
        # map, where one exists, is better than either.
        #
        # Procedural relief goes on objects only. A roughness map is invisible on a
        # wall, but relief is not: our maps cover UV [0, 1] once with no tiling, so on
        # a large structural surface the bumps are stretched to the size of the wall
        # and read as blotchy mottling rather than texture. Objects are small and their
        # UV atlases dense, which is the case the maps are sized for. Authored maps are
        # exempt -- they were drawn for the surface they belong to.
        normal = _load_map(real["normal"], noise_resolution) if "normal" in real else None
        if normal is not None:
            stem = albedo_stem(
                albedo_files.get(material.textures[int(mj.mjtTextureRole.mjTEXROLE_RGB)], "")
            )
            material.textures[int(mj.mjtTextureRole.mjTEXROLE_NORMAL)] = authored_texture(
                stem, "normal", normal[..., :3]
            )
            stats.authored_normal += 1
        elif bump_strength > 0.0 and finish != "flat" and source == "from_metadata":
            material.textures[int(mj.mjtTextureRole.mjTEXROLE_NORMAL)] = normal_for(finish)
            stats.procedural_normal += 1

        setattr(stats, source, getattr(stats, source) + 1)
        stats.finishes[finish] = stats.finishes.get(finish, 0) + 1
        if pbr_class.metallic == 1.0:
            stats.metals += 1

    log.info(
        "PBR overrides: %d/%d materials (%d from metadata, %d from name, %d fallback, "
        "%d metal; %d real normal maps, %d real metallic/roughness, %d procedural "
        "relief); skipped %d already authored, %d translucent",
        stats.overridden,
        stats.total,
        stats.from_metadata,
        stats.from_name,
        stats.fallback,
        stats.metals,
        stats.authored_normal,
        stats.authored_metallic_roughness,
        stats.procedural_normal,
        stats.skipped_authored,
        stats.skipped_transparent,
    )
    log.debug("PBR finishes drawn: %s", dict(sorted(stats.finishes.items())))
    return stats

    def classify_composition(materials: list[str], composition: list[float]) -> PbrClass:
    """Pick a PBR class from an objathor ``materials``/``composition`` annotation.

    ``composition`` is a volume fraction, not a surface fraction, so this is a proxy:
    it takes the largest non-filler component as the surface, then lets any
    meaningful coat of paint or glaze override a metal verdict.
    """
    if not materials:
        return UNKNOWN

    tokens = [str(m).strip().lower() for m in materials]
    if not composition or len(composition) != len(tokens):
        # Annotations always agree in the shipped metadata, but do not trust it blindly.
        fractions = [1.0 / len(tokens)] * len(tokens)
    else:
        fractions = [float(c) for c in composition]

    by_token: dict[str, float] = {}
    for token, fraction in zip(tokens, fractions, strict=True):
        by_token[token] = by_token.get(token, 0.0) + fraction

    surface = {t: f for t, f in by_token.items() if t not in _NON_SURFACE}
    if not surface:
        return UNKNOWN

    dominant = max(surface, key=lambda t: surface[t])
    chosen = lookup_token(dominant) or UNKNOWN

    if chosen.metallic == 1.0:
        coat = max(
            (t for t in _COATINGS if surface.get(t, 0.0) >= _COATING_FRACTION),
            key=lambda t: surface[t],
            default=None,
        )
        if coat is not None:
            return MATERIAL_CLASSES[coat]

    return chosen

def finish_field(finish: str, resolution: int = 1024, seed: int = 0) -> np.ndarray:
    """The normalised height field behind a finish, in [-1, 1] with zero mean.

    Both maps a finish produces come from this one field, so a surface's roughness
    variation and its relief describe the same physical texture: where the weave
    stands proud it is also glossier, because that is the thread the light catches.
    Generating them independently would put the two out of register and read as two
    unrelated patterns laid over each other.
    """
    if finish not in ROUGHNESS_FINISHES:
        raise KeyError(f"unknown finish {finish!r}; known: {sorted(ROUGHNESS_FINISHES)}")

    rng = np.random.default_rng(seed)
    field = ROUGHNESS_FINISHES[finish](resolution, rng)

    # Trim the tail first -- several finishes are skewed rather than symmetric, and
    # clipping a skewed field is not mean-preserving -- and only then centre and
    # normalise. Recentring *after* the clip is what gives the exact zero mean that
    # the roughness multiply depends on; scaling by the peak rather than the standard
    # deviation makes the caller's amplitude the true maximum deviation.
    field = field - field.mean()
    std = float(field.std())
    if std > 1e-6:
        field = field / std
        field = np.clip(field, -3.0, 3.0)
        field = field - field.mean()
        peak = float(np.abs(field).max())
        if peak > 1e-6:
            field = field / peak
    field.flags.writeable = False
    return field

def normal_texture(
    finish: str = "smudged",
    resolution: int = 1024,
    strength: float = 0.10,
    seed: int = 0,
) -> np.ndarray:
    """Build a tangent-space normal map from a finish's height field.

    Roughness and relief are not the same thing. Roughness is a statistic of facets
    below the texel, and only ever blurs or sharpens a reflection -- if the reflected
    environment is dim, as ours are, it barely shows. A normal map instead turns the
    surface to face a different way at every texel, so each bump gets a lit side and a
    shaded side and produces luminance contrast directly. Measured on a macro shot of
    one object, a normal map at strength 0.35 moved 82% of pixels where the roughness
    map at the same framing moved 6%, so this is the knob that carries visible detail
    -- and the reason the default here is deliberately gentle.

    ``strength`` is the RMS surface slope, which keeps it independent of resolution:
    0.10 is a soft texture, 0.35 is aggressively embossed.

    Returns HxWx3 uint8 in the usual encoding, ``rgb = normal * 0.5 + 0.5``, so a flat
    surface is (128, 128, 255). It must be uploaded as linear, not sRGB.
    """
    field = finish_field(finish, resolution, seed)

    # Gradient of the height field; the surface normal is (-dh/du, -dh/dv, 1).
    grad_v, grad_u = np.gradient(field.astype(np.float32))
    rms = float(np.sqrt(np.mean(grad_u**2 + grad_v**2)))
    if rms > 1e-9:
        scale = strength / rms
        grad_u = grad_u * scale
        grad_v = grad_v * scale

    normal = np.stack([-grad_u, -grad_v, np.ones_like(grad_u)], axis=-1)
    normal /= np.linalg.norm(normal, axis=-1, keepdims=True)

    texture = np.clip((normal * 0.5 + 0.5) * 255.0, 0.0, 255.0).astype(np.uint8)
    texture.flags.writeable = False
    return texture
